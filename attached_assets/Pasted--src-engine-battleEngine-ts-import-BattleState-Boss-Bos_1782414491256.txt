// src/engine/battleEngine.ts

import {
  BattleState,
  Boss,
  BossRespawnState
} from "../types/game";

export const ONE_SECOND = 1000;
export const ONE_MINUTE = 60 * ONE_SECOND;
export const ONE_HOUR = 60 * ONE_MINUTE;
export const ONE_DAY = 24 * ONE_HOUR;

export const CRITICAL_CHANCE = 0.1;
export const CRITICAL_MULTIPLIER = 2;

export function calculateBattleTime(
  bossHp: number,
  dps: number
) {
  if (dps <= 0) {
    return {
      totalSeconds: Infinity,
      days: 0,
      hours: 0,
      minutes: 0,
      seconds: 0
    };
  }

  const totalSeconds = bossHp / dps;

  return {
    totalSeconds,
    days: Math.floor(totalSeconds / 86400),
    hours: Math.floor((totalSeconds % 86400) / 3600),
    minutes: Math.floor((totalSeconds % 3600) / 60),
    seconds: Math.floor(totalSeconds % 60)
  };
}

export function formatTimeLeft(
  milliseconds: number
): string {
  if (milliseconds <= 0) {
    return "00:00:00";
  }

  const totalSeconds = Math.floor(
    milliseconds / 1000
  );

  const days = Math.floor(
    totalSeconds / 86400
  );

  const hours = Math.floor(
    (totalSeconds % 86400) / 3600
  );

  const minutes = Math.floor(
    (totalSeconds % 3600) / 60
  );

  const seconds = totalSeconds % 60;

  return `${days}д ${hours}ч ${minutes}м ${seconds}с`;
}

export function calculateRealDps(
  baseDps: number,
  boost20: boolean,
  x2Boost: boolean
): number {
  let result = baseDps;

  if (boost20) {
    result *= 1.2;
  }

  if (x2Boost) {
    result *= 2;
  }

  return result;
}

export function rollCriticalHit(
  damage: number
) {
  const critical =
    Math.random() <= CRITICAL_CHANCE;

  return {
    critical,
    damage: critical
      ? damage * CRITICAL_MULTIPLIER
      : damage
  };
}

export function createBattle(
  boss: Boss,
  totalDps: number
): BattleState {
  const now = Date.now();

  const battleDuration =
    (boss.maxHp / totalDps) * 1000;

  return {
    active: true,

    bossId: boss.id,

    bossCurrentHp: boss.maxHp,

    bossMaxHp: boss.maxHp,

    totalDps,

    startedAt: now,

    estimatedEndAt:
      now + battleDuration,

    boost20Active: false,

    boost20ExpiresAt: null,

    x2BoostActive: false,

    x2BoostExpiresAt: null
  };
}

export function getCurrentBossHp(
  battle: BattleState
): number {
  if (!battle.active) {
    return 0;
  }

  const now = Date.now();

  const elapsedSeconds =
    (now - battle.startedAt) / 1000;

  const damageDone =
    elapsedSeconds * battle.totalDps;

  const hp =
    battle.bossMaxHp - damageDone;

  return Math.max(0, hp);
}

export function getBattleProgress(
  battle: BattleState
): number {
  const hp = getCurrentBossHp(battle);

  const progress =
    ((battle.bossMaxHp - hp) /
      battle.bossMaxHp) *
    100;

  return Math.min(
    100,
    Math.max(0, progress)
  );
}

export function isBattleFinished(
  battle: BattleState
): boolean {
  return (
    getCurrentBossHp(battle) <= 0
  );
}

export function getRemainingTime(
  battle: BattleState
): number {
  const now = Date.now();

  return Math.max(
    0,
    battle.estimatedEndAt - now
  );
}

export function finishBattle(
  battle: BattleState
): BattleState {
  return {
    ...battle,
    active: false,
    bossCurrentHp: 0
  };
}

export function createRespawnTimer(): BossRespawnState {
  return {
    available: false,

    respawnAt:
      Date.now() + ONE_DAY,

    adsWatched: 0
  };
}

export function isBossAvailable(
  respawn: BossRespawnState
): boolean {
  if (respawn.available) {
    return true;
  }

  if (!respawn.respawnAt) {
    return true;
  }

  return Date.now() >= respawn.respawnAt;
}

export function getRespawnTimeLeft(
  respawn: BossRespawnState
): number {
  if (
    !respawn.respawnAt ||
    respawn.available
  ) {
    return 0;
  }

  return Math.max(
    0,
    respawn.respawnAt - Date.now()
  );
}

export function addRespawnAdView(
  respawn: BossRespawnState,
  adsRequired: number
): BossRespawnState {
  const watched =
    respawn.adsWatched + 1;

  if (watched >= adsRequired) {
    return {
      available: true,
      respawnAt: null,
      adsWatched: watched
    };
  }

  return {
    ...respawn,
    adsWatched: watched
  };
}

export function applyFinishDamage(
  currentHp: number,
  maxHp: number
): number {
  const damage =
    maxHp * 0.01;

  return Math.max(
    0,
    currentHp - damage
  );
}

export function getFinishPercent(
  currentHp: number,
  maxHp: number
): number {
  return (
    (currentHp / maxHp) * 100
  );
}

export function calculateStarsFinishCost(
  currentHp: number,
  maxHp: number
): number {
  const percentLeft =
    currentHp / maxHp;

  return Math.ceil(
    percentLeft * 200
  );
}

export function calculateTonFinishCost(
  currentHp: number,
  maxHp: number
): number {
  const percentLeft =
    currentHp / maxHp;

  return Number(
    (percentLeft * 1).toFixed(2)
  );
}

export function saveBattleState(
  battle: BattleState
) {
  localStorage.setItem(
    "boss_hunt_battle",
    JSON.stringify(battle)
  );
}

export function loadBattleState():
  | BattleState
  | null {
  const raw =
    localStorage.getItem(
      "boss_hunt_battle"
    );

  if (!raw) {
    return null;
  }

  try {
    return JSON.parse(raw);
  } catch {
    return null;
  }
}

export function clearBattleState() {
  localStorage.removeItem(
    "boss_hunt_battle"
  );
}
